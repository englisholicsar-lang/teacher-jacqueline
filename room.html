function startVapiCall() { if (window.vapiSDK && typeof window.vapiSDK.run === 'function') { vapiInstance = window.vapiSDK.run({ apiKey: VAPI_KEY, assistant: activeAssistantId, config: {} }); if (vapiInstance && typeof vapiInstance.on === 'function') { vapiInstance.on('call-start', () => { startTimer(); const remoteAudio = document.getElementById("remote-audio"); if (remoteAudio) { remoteAudio.muted = false; remoteAudio.volume = 1.0; remoteAudio.play().catch(e => console.log("Audio play error:", e)); } }); vapiInstance.on('speech-start', () => { if (!timerInterval) startTimer(); }); vapiInstance.on('track', (track) => { if (track.kind === 'audio') { const remoteAudio = document.getElementById("remote-audio"); if (remoteAudio) { remoteAudio.muted = false; remoteAudio.srcObject = new MediaStream([track]); remoteAudio.volume = 1.0; remoteAudio.play().then(() => { console.log("Remote audio stream playing successfully"); }).catch(e => console.log("Stream play error:", e)); } } }); vapiInstance.on('call-end', () => { endCallUI(); }); } setTimeout(() => { if (callState === 'IDLE') { startTimer(); callState = 'ACTIVE'; document.getElementById("call-status").innerText = `${sessionStudentName} 회원님, 통화가 연결되었습니다.`; document.getElementById("notice-box").innerHTML = `
🎙️
선생님께 인사해 보세요! (예: Hello!)
`; const btn = document.getElementById("action-btn"); btn.style.backgroundColor = "#e05252"; btn.style.boxShadow = "0 6px 16px rgba(224, 82, 82, 0.35)"; btn.removeAttribute("disabled"); document.getElementById("btn-text-content").innerText = "통화 종료하기"; } }, 1500); } else { alert("Vapi SDK 로딩 중입니다. 잠시 후 다시 시도해 주세요."); const btn = document.getElementById("action-btn"); btn.removeAttribute("disabled"); document.getElementById("btn-text-content").innerText = "📞 통화시작하기"; } }
