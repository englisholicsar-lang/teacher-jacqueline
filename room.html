async function handleMainAction() { if (!isUserAllowed) return; const remoteAudio = document.getElementById("remote-audio"); if (remoteAudio) { remoteAudio.volume = 1.0; remoteAudio.play().catch(e => console.log("Audio unlock tap", e)); } if (callState === 'IDLE') { const btn = document.getElementById("action-btn"); btn.setAttribute("disabled", "true"); document.getElementById("btn-text-content").innerText = "마이크 권한 요청 중..."; try { // Explicitly trigger the browser's native microphone permission prompt const micStream = await navigator.mediaDevices.getUserMedia({ audio: true }); window.activeMicStream = micStream; // Keep track of stream document.getElementById("btn-text-content").innerText = "통화 연결 중..."; startVapiCall(); } catch (err) { console.error("Microphone permission denied:", err); alert("마이크 권한이 거부되었거나 지원되지 않는 환경입니다. 브라우저 설정에서 마이크를 허용해 주세요."); btn.removeAttribute("disabled"); document.getElementById("btn-text-content").innerText = "📞 통화시작하기"; } } else if (callState === 'ACTIVE') { endCallUI(); } }
