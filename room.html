async function handleMainAction() { if (!isUserAllowed) return; const remoteAudio = document.getElementById("remote-audio"); if (remoteAudio) { remoteAudio.volume = 1.0; remoteAudio.play().catch(e => console.log("Audio unlock tap", e)); } if (callState === 'IDLE') { const btn = document.getElementById("action-btn"); btn.setAttribute("disabled", "true"); document.getElementById("btn-text-content").innerText = "마이크 권한 요청 중..."; try { const micStream = await navigator.mediaDevices.getUserMedia({ audio: true }); window.activeMicStream = micStream; document.getElementById("btn-text-content").innerText = "통화 연결 중..."; // 1. Initialize Vapi instance if (window.vapiSDK && typeof window.vapiSDK.run === 'function') { vapiInstance = window.vapiSDK.run({ apiKey: VAPI_KEY, assistant: activeAssistantId, config: {} }); // 2. Attach event listeners if (vapiInstance && typeof vapiInstance.on === 'function') { vapiInstance.on('call-start', () => { startTimer(); }); vapiInstance.on('track', (track) => { if (track.kind === 'audio') { const remoteAudio = document.getElementById("remote-audio"); if (remoteAudio) { remoteAudio.muted = false; remoteAudio.srcObject = new MediaStream([track]); remoteAudio.volume = 1.0; remoteAudio.play().catch(e => console.log("Stream play error:", e)); } } }); vapiInstance.on('call-end', () => { endCallUI(); }); } // 3. EXPLICITLY TRIGGER VAPI'S CALL START if (vapiInstance && typeof vapiInstance.start === 'function') { vapiInstance.start(activeAssistantId); } } // 4. Transition UI to Active Call State startTimer(); callState = 'ACTIVE'; document.getElementById("call-status").innerText = `${sessionStudentName} 회원님, 통화가 연결되었습니다.`; document.getElementById("notice-box").innerHTML = `
🎙️
선생님께 인사해 보세요! (예: Hello!)
`; btn.style.backgroundColor = "#e05252"; btn.style.boxShadow = "0 6px 16px rgba(224, 82, 82, 0.35)"; btn.removeAttribute("disabled"); document.getElementById("btn-text-content").innerText = "통화 종료하기"; } catch (err) { console.error("Microphone permission denied:", err); alert("마이크 권한이 거부되었거나 지원되지 않는 환경입니다. 브라우저 설정에서 마이크를 허용해 주세요."); btn.removeAttribute("disabled"); document.getElementById("btn-text-content").innerText = "📞 통화시작하기"; } } else if (callState === 'ACTIVE') { endCallUI(); } }
